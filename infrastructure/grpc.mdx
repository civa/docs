```
https://grpc.solami.fast?api_key=YOUR_API_KEY
```

## Features

- Yellowstone gRPC protocol
- Subscribe to accounts, transactions, blocks, and slots
- Filter by program, account, or transaction type
- Zero missed events with reliable delivery
- Available on Pro plan and above

## Usage Example

```rust
#![allow(deprecated)]
use futures_util::StreamExt;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let _ = dotenvy::dotenv();

    let mut client = solami::from_env().build().await?;

    let accounts = vec![
        "11111111111111111111111111111111".to_string(),
    ];

    let (_sink, mut stream) = client.grpc().subscribe_transactions(
        "tx_sub",
        accounts,
        solami::CommitmentLevel::Confirmed,
    ).await?;

    println!("Subscribed to transactions via gRPC...");

    while let Some(Ok(msg)) = stream.next().await {
        if let Some(solami::GrpcUpdateKind::Transaction(tx)) = msg.update_oneof {
            if let Some(info) = tx.transaction {
                println!("Signature: {:?} | Slot: {}", info.signature, tx.slot);
            }
        }
    }

    Ok(())
}
```
